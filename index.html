    function runAnalysis() {

        const date = document.getElementById('analysisDate').value;

        if (!date) return showToast("Pilih tarikh dahulu", "warning");


        const records = allAttendanceRecords.filter(r => r.date === date);

        const formFilter = document.getElementById('absentFormFilter').value;


        // Stats

        // Simple logic: Unique students who have at least one record today

        const uniqueStudentsToday = [...new Set(records.map(r => r.student_id))];

        const studentStats = {};

        

        // Determine if student is fully present or has any absence

        uniqueStudentsToday.forEach(sid => {

            const sRecords = records.filter(r => r.student_id === sid);

            const hasAbsence = sRecords.some(r => r.status === 'TIDAK HADIR');

            studentStats[sid] = hasAbsence ? 'ABSENT' : 'PRESENT';

        });


        const presentCount = Object.values(studentStats).filter(s => s === 'PRESENT').length;

        const absentCount = Object.values(studentStats).filter(s => s === 'ABSENT').length;

        

        document.getElementById('totalRegistered').innerText = allStudents.length;

        document.getElementById('totalPresent').innerText = presentCount;

        document.getElementById('totalAbsent').innerText = absentCount;

        const pct = (presentCount + absentCount) > 0 ? (presentCount / (presentCount + absentCount) * 100).toFixed(1) : 0;

        document.getElementById('attendancePercent').innerText = `${pct}%`;


        // Absent List

        let absentListHTML = '';

        let absentRecords = records.filter(r => r.status === 'TIDAK HADIR');

        if (formFilter) absentRecords = absentRecords.filter(r => r.form == formFilter);


        // Deduplicate: show one row per student per subject or just list them? 

        // Let's list specific subject absences

        if (absentRecords.length === 0) {

            absentListHTML = '<div class="p-4 text-center text-green-600 bg-green-50 rounded">Tiada ketidakhadiran direkodkan!</div>';

            document.getElementById('downloadAnalysisPdfBtn').classList.add('hidden');

        } else {

            document.getElementById('downloadAnalysisPdfBtn').classList.remove('hidden');

            absentListHTML = absentRecords.map(r => `

                <div class="absent-row flex items-center justify-between p-3 border-b last:border-0 hover:bg-red-50" 

                     data-name="${r.name}" data-class="${r.form} ${r.class_name}" data-reason="${r.reason}">

                    <div>

                        <p class="font-bold text-gray-800">${r.name}</p>

                        <p class="text-xs text-gray-500">Tingkatan ${r.form} ${r.class_name} • ${r.subject}</p>

                    </div>

                    <span class="px-2 py-1 bg-red-100 text-red-800 text-xs font-bold rounded">${r.reason || 'Tiada Sebab'}</span>

                </div>

            `).join('');

        }

        document.getElementById('absentStudentsList').innerHTML = absentListHTML;

    }
